
using Godot;

public partial class Lobby : Control
{
    private NetworkManager _networkManager;

    private PopupPanel _hostPopup;
    private PopupPanel _joinPopup;

    private Control _hostContent;
    private Control _joinContent;

    private Button _hostButton;
    private Button _joinButton;
    private Button _backButton;

    private Button _hostCancelButton;
    private Button _joinCancelButton;
    private Button _createRoomButton;

    private LineEdit _hostNameInput;
    private LineEdit _joinNameInput;
    private LineEdit _joinCodeInput;

    private string _playerName = "";
    private string _lastSearchedCode = "";

    private Timer _searchDebounceTimer;
    private const double SearchDelay = 0.6;

    public override void _Ready()
    {
        // Network manager
        _networkManager =
            GetNodeOrNull<NetworkManager>("/root/NetworkManager");

        // Main lobby buttons
        _hostButton =
            GetNodeOrNull<Button>("HostPanel/btnHost");

        _joinButton =
            GetNodeOrNull<Button>("JoinPanel/btnJoin");

        _backButton =
            GetNodeOrNull<Button>("Back");

        // Popups
        _hostPopup =
            GetNodeOrNull<PopupPanel>("HostPopup");

        _joinPopup =
            GetNodeOrNull<PopupPanel>("JoinPopup");

        // Popup content
        _hostContent =
            GetNodeOrNull<Control>("HostPopup/HostContent");

        _joinContent =
            GetNodeOrNull<Control>("JoinPopup/JoinContent");

        _roomResultControl =
        _joinContent.FindChild("playerRoom", true, false) as Control;

        _foundRoomStyle = new StyleBoxFlat();
        _foundRoomStyle.BgColor = new Color(0.08f, 0.12f, 0.08f);
        _foundRoomStyle.BorderColor = new Color(0.15f, 0.9f, 0.3f);
        _foundRoomStyle.SetBorderWidthAll(3);
        _foundRoomStyle.SetCornerRadiusAll(8);

        // Check required nodes before using them
        if (_networkManager == null ||
            _hostButton == null ||
            _joinButton == null ||
            _backButton == null ||
            _hostPopup == null ||
            _joinPopup == null ||
            _hostContent == null ||
            _joinContent == null)
        {
            GD.PrintErr(
                "Lobby: A required node is missing. " +
                "Check the node names and paths in the scene tree."
            );
            return;
        }

        // Find inputs, including inputs nested inside containers
        _hostNameInput =
            FindLineEdit(_hostContent, "inputName");

        _joinNameInput =
            FindLineEdit(_joinContent, "inputName");

        _joinCodeInput =
            FindLineEdit(_joinContent, "InputCode");


        // Check important input nodes
        if (_hostNameInput == null)
            GD.PrintErr("HostPopup: inputName was not found.");

        if (_joinNameInput == null)
            GD.PrintErr("JoinPopup: inputName was not found.");

        if (_joinCodeInput == null)
            GD.PrintErr("JoinPopup: InputCode was not found.");

        // Connect player-name synchronization AFTER finding inputs
        if (_hostNameInput != null)
            _hostNameInput.TextChanged += OnHostNameChanged;

        if (_joinNameInput != null)
            _joinNameInput.TextChanged += OnJoinNameChanged;

        // Search when the player presses Enter or leaves InputCode
        if (_joinCodeInput != null)
        {
            _joinCodeInput.FocusExited += OnRoomCodeFinished;
            _joinCodeInput.TextSubmitted += OnRoomCodeSubmitted;
            _joinCodeInput.TextChanged += OnRoomCodeChanged;
        }

        _searchDebounceTimer = new Timer();
        _searchDebounceTimer.WaitTime = SearchDelay;
        _searchDebounceTimer.OneShot = true;
        AddChild(_searchDebounceTimer);

        _searchDebounceTimer.Timeout += SearchRoomFromInput;

        if (_joinCodeInput != null)
        {
            _joinCodeInput.TextChanged += OnRoomCodeChanged;
        }

        // Hide popups when the lobby opens
        _hostPopup.Hide();
        _joinPopup.Hide();

        // Connect main buttons
        _hostButton.Pressed += ShowHostPopup;
        _joinButton.Pressed += ShowJoinPopup;
        _backButton.Pressed += OnBackButtonPressed;

        // Cancel buttons
        _hostCancelButton =
            FindButton(_hostContent, "btnCancel");

        _joinCancelButton =
            FindButton(_joinContent, "btnCancel");

        if (_hostCancelButton != null)
            _hostCancelButton.Pressed += OnCancelPopup;

        if (_joinCancelButton != null)
            _joinCancelButton.Pressed += OnCancelPopup;

        // Host: Create room button
        _createRoomButton =
            FindButton(_hostContent, "btnRoom");

        if (_createRoomButton != null)
            _createRoomButton.Pressed += OnCreateRoomPressed;
        else
            GD.PrintErr("HostPopup: btnRoom was not found.");

        // Listen for room search results
        _networkManager.RoomFound += OnRoomFound;
        _networkManager.RoomSearchFailed += OnRoomSearchFailed;

        GD.Print("Lobby loaded successfully.");
    }

    // Find a LineEdit anywhere inside a popup's content.
    private LineEdit FindLineEdit(Control root, string nodeName)
    {
        if (root == null)
            return null;

        return root.FindChild(nodeName, true, false) as LineEdit;
    }

    // Find a Button anywhere inside a popup's content.
    private Button FindButton(Control root, string nodeName)
    {
        if (root == null)
            return null;

        return root.FindChild(nodeName, true, false) as Button;
    }

    // Update the player name in both popups.
    private void OnHostNameChanged(string newText)
    {
        _playerName = newText;

        if (_joinNameInput != null &&
            _joinNameInput.Text != newText)
        {
            _joinNameInput.Text = newText;
        }
    }

    private void OnJoinNameChanged(string newText)
    {
        _playerName = newText;

        if (_hostNameInput != null &&
            _hostNameInput.Text != newText)
        {
            _hostNameInput.Text = newText;
        }
    }

    // Show Host popup.
    private void ShowHostPopup()
    {
        _joinPopup.Hide();

        if (_hostNameInput != null)
            _hostNameInput.Text = _playerName;

        _hostPopup.PopupCentered();
    }

    // Show Join popup.
    private void ShowJoinPopup()
    {
        _hostPopup.Hide();

        if (_joinNameInput != null)
            _joinNameInput.Text = _playerName;

        _joinPopup.PopupCentered();
    }

    // Close both popups.
    private void OnCancelPopup()
    {
        _hostPopup.Hide();
        _joinPopup.Hide();
    }

   
    // Return to the main menu.
    private void OnBackButtonPressed()
    {
        GetTree().ChangeSceneToFile("res://scenes/menu.tscn");
    }

    // HOST: Create a room.
    private void OnCreateRoomPressed()
    {
        if (_networkManager == null || _hostContent == null)
        {
            GD.PrintErr("Cannot create room: NetworkManager or HostContent is missing.");
            return;
        }

        LineEdit playerNameInput =
            FindLineEdit(_hostContent, "inputName");

        LineEdit roomNameInput =
            FindLineEdit(_hostContent, "inputRoomName");

        Node roomCodeNode =
            _hostContent.FindChild("txtGeneratedRoomCode", true, false);

        // Verify the required input nodes exist.
        if (playerNameInput == null || roomNameInput == null)
        {
            GD.PrintErr(
                "Cannot create room. Check these node names inside HostContent: " +
                "inputName and inputRoomName."
            );

            GD.PrintErr(
                $"inputName found: {playerNameInput != null}; " +
                $"inputRoomName found: {roomNameInput != null}"
            );
            return;
        }

        // Read and trim the actual input values.
        string playerName = playerNameInput.Text.Trim();
        string roomName = roomNameInput.Text.Trim();

        // Debug output: shows exactly what the game reads.
        GD.Print($"Player name entered: '{playerName}'");
        GD.Print($"Room name entered: '{roomName}'");

        // Do not call NetworkManager if either field is empty.
        if (string.IsNullOrWhiteSpace(playerName) ||
            string.IsNullOrWhiteSpace(roomName))
        {
            GD.PrintErr(
                "Enter both a player name and a room name."
            );
            return;
        }

        // Create the room.
        Error result =
            _networkManager.CreateRoom(playerName, roomName);

        if (result != Error.Ok)
        {
            SetNodeText(roomCodeNode, "");
            GD.PrintErr($"Room creation failed: {result}");
            return;
        }

        // Show the generated room code.
        if (roomCodeNode is LineEdit codeInput)
            codeInput.Editable = false;

        SetNodeText(roomCodeNode, _networkManager.RoomCode);

        GD.Print(
            $"Room created successfully! " +
            $"Room: {_networkManager.RoomName}, " +
            $"Code: {_networkManager.RoomCode}"
        );
    }

    // JOIN: Search using the code typed into InputCode.
    private void SearchRoomFromInput()
    {
        if (_networkManager == null || _joinCodeInput == null)
            return;

        string code =
            _joinCodeInput.Text.Trim().ToUpperInvariant();

        if (string.IsNullOrWhiteSpace(code))
        {
            SetRoomDisplay("Enter a room code first.");
            return;
        }

        // Avoid searching repeatedly for the same code.
        if (code == _lastSearchedCode)
            return;

        _lastSearchedCode = code;

        SetRoomDisplay("Searching for room...");
        GD.Print($"Searching for room code: {code}");

        _networkManager.SearchForRoom(code);
    }

    private void OnRoomCodeFinished()
    {
        SearchRoomFromInput();
    }

    private void OnRoomCodeSubmitted(string text)
    {
        SearchRoomFromInput();
    }

    private void OnRoomCodeChanged(string newText)
    {
        SetRoomFoundOutline(false);
        _lastSearchedCode = "";

        if (_searchDebounceTimer == null)
            return;

        _searchDebounceTimer.Stop();

        string code = newText.Trim().ToUpperInvariant();

        if (string.IsNullOrWhiteSpace(code))
        {
            SetRoomDisplay("");
            return;
        }

        SetRoomDisplay("Searching for room...");
        _searchDebounceTimer.Start();
    }

    // Display a successful room search.
    private void OnRoomFound(
        string roomName,
        string hostName,
        string hostIP,
        int roomPort)
    {
        SetRoomDisplay(
            $"Room found: {roomName} | Host: {hostName}"
        );

        GD.Print(
            $"Room found! Name: {roomName}, " +
            $"Host: {hostName}, IP: {hostIP}, Port: {roomPort}"
        );

        SetRoomFoundOutline(true);
    }

    // Display a failed room search.
    private void OnRoomSearchFailed(string message)
    {
        SetRoomDisplay(message);
        GD.PrintErr(message);
        SetRoomFoundOutline(false);
    }

    // Update the playerRoom result display.
    private void SetRoomDisplay(string message)
    {
        if (_joinContent == null)
            return;

        Node display =
            _joinContent.FindChild("playerRoom", true, false);

        SetNodeText(display, message);
    }

    // Set text on common Godot text controls.
    private void SetNodeText(Node node, string message)
    {
        switch (node)
        {
            case Label label:
                label.Text = message;
                break;

            case LineEdit lineEdit:
                lineEdit.Text = message;
                break;

            case RichTextLabel richTextLabel:
                richTextLabel.Text = message;
                break;

            case TextEdit textEdit:
                textEdit.Text = message;
                break;

            case null:
                GD.PrintErr("Could not find the requested text node.");
                break;

            default:
                GD.PrintErr(
                    $"Unsupported text node type: {node.GetType().Name}"
                );
                break;
        }
    }


    private Control _roomResultControl;
    private StyleBoxFlat _foundRoomStyle;

    private void SetRoomFoundOutline(bool found)
    {
        if (_roomResultControl == null)
            return;

        if (_roomResultControl is LineEdit lineEdit)
        {
            if (found)
            {
                lineEdit.AddThemeStyleboxOverride("normal", _foundRoomStyle);
                lineEdit.AddThemeStyleboxOverride("focus", _foundRoomStyle);
            }
            else
            {
                lineEdit.RemoveThemeStyleboxOverride("normal");
                lineEdit.RemoveThemeStyleboxOverride("focus");
            }
        }
    }


    public override void _ExitTree()
    {
        // Disconnect player-name signals.
        if (_hostNameInput != null &&
            GodotObject.IsInstanceValid(_hostNameInput))
        {
            _hostNameInput.TextChanged -= OnHostNameChanged;
        }

        if (_joinNameInput != null &&
            GodotObject.IsInstanceValid(_joinNameInput))
        {
            _joinNameInput.TextChanged -= OnJoinNameChanged;
        }

        // Disconnect room-code signals.
        if (_joinCodeInput != null &&
            GodotObject.IsInstanceValid(_joinCodeInput))
        {
            _joinCodeInput.FocusExited -= OnRoomCodeFinished;
            _joinCodeInput.TextSubmitted -= OnRoomCodeSubmitted;
            _joinCodeInput.TextChanged -= OnRoomCodeChanged;
        }

        // Disconnect NetworkManager signals.
        if (_networkManager != null &&
            GodotObject.IsInstanceValid(_networkManager))
        {
            _networkManager.RoomFound -= OnRoomFound;
            _networkManager.RoomSearchFailed -= OnRoomSearchFailed;
        }

        // Disconnect main buttons.
        if (_hostButton != null &&
            GodotObject.IsInstanceValid(_hostButton))
        {
            _hostButton.Pressed -= ShowHostPopup;
        }

        if (_joinButton != null &&
            GodotObject.IsInstanceValid(_joinButton))
        {
            _joinButton.Pressed -= ShowJoinPopup;
        }

        if (_backButton != null &&
            GodotObject.IsInstanceValid(_backButton))
        {
            _backButton.Pressed -= OnBackButtonPressed;
        }

        // Disconnect cancel and create buttons.
        if (_hostCancelButton != null &&
            GodotObject.IsInstanceValid(_hostCancelButton))
        {
            _hostCancelButton.Pressed -= OnCancelPopup;
        }

        if (_joinCancelButton != null &&
            GodotObject.IsInstanceValid(_joinCancelButton))
        {
            _joinCancelButton.Pressed -= OnCancelPopup;
        }

        if (_createRoomButton != null &&
            GodotObject.IsInstanceValid(_createRoomButton))
        {
            _createRoomButton.Pressed -= OnCreateRoomPressed;
        }
    }
}
